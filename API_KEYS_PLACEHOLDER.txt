## 1. 后端配置文件: backend/app/config.py
##    路径: backend/app/config.py
##    查找所有 "chenxxxxxx" 占位符并替换

[数据库]
DATABASE_URL = postgresql+asyncpg://USER:PASSWORD@localhost:5432/fengdiao_phm
# 替换: USER, PASSWORD

[Redis]
REDIS_URL = redis://localhost:6379/0
# 如需密码: redis://:PASSWORD@localhost:6379/0

[JWT密钥]
SECRET_KEY = <生成一个随机字符串>
# 生成方式: python -c "import secrets; print(secrets.token_hex(32))"

[邮件SMTP]
SMTP_HOST = smtp.example.com          # 替换为你的SMTP服务器
SMTP_PORT = 587
SMTP_USER = your-email@example.com    # 替换为你的邮箱
SMTP_PASSWORD = your-email-password   # 替换为邮箱密码/授权码
SMTP_FROM = your-email@example.com    # 替换为发件人地址

[ERA5气象数据]
ERA5_CDS_API_KEY = <your-cds-api-key>
ERA5_CDS_API_URL = https://cds-beta.climate.copernicus.eu/api
# 注册地址: https://cds-beta.climate.copernicus.eu/
# 获取API Key: https://cds-beta.climate.copernicus.eu/how-to-api

[地图服务]
MAPBOX_TOKEN = <your-mapbox-token>
# 如使用Mapbox: https://account.mapbox.com/
# 当前项目使用免费的Leaflet+OpenStreetMap，此项可选

[AI大模型API]
OPENAI_API_KEY = sk-xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx
OPENAI_BASE_URL = https://api.openai.com/v1
# 或使用其他兼容API (如DeepSeek, 通义千问等)

[企业微信通知]
WEIXIN_WEBHOOK_URL = https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=xxxxxxxx
# 获取方式: 企业微信管理后台 → 应用管理 → 群机器人
